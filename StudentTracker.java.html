1    import javax.swing.JOptionPane;
2    
3    /**
4     * Driver class for the Student Tracker program.
5     */
6    public class StudentTracker
7    {
8        public static void main(String[] args)
9        {
10           Tracker tracker = new Tracker();
11   
12           while (true)
13           {
14               String menu =
15                       "Welcome to Student Tracker\n\n" +
16                               "1. Add Student\n" +
17                               "2. Mark Attendance\n" +
18                               "3. Add Lesson\n" +
19                               "4. View All Students\n" +
20                               "5. Save to File\n" +
21                               "6. Load from File\n" +
22                               "7. Delete Student\n" +
23                               "8. View One Student\n" +
24                               "9. Edit Student\n" +
25                               "10. Exit";
26               String input = JOptionPane.showInputDialog(menu);
27               if (input == null) break;
28   
29               int choice;
30               try// utilized a try catch because crashing shouldn't occur if user enters the wrong type of input (string instead of int for example)
31               {
32                   choice = Integer.parseInt(input);
33               }
34               catch (NumberFormatException e) //prevents the StudentTracker from crashing by showing invalid input
35               {
36                   JOptionPane.showMessageDialog(null, "Invalid input.");
37                   continue;
38               }
39   
40               switch (choice)
41               {
42                   case 1:
43                       tracker.addStudent(
44                               JOptionPane.showInputDialog("First name:"),
45                               JOptionPane.showInputDialog("Last name:"),
46                               Integer.parseInt(
47                                       JOptionPane.showInputDialog("Grade level:"))
48                       );
49                       break;
50   
51                   case 2:
52                       tracker.markAttendance(
53                               JOptionPane.showInputDialog("First name:"),
54                               JOptionPane.showInputDialog("Last name:")
55                       );
56                       break;
57   
58                   case 3:
59                       tracker.addCompletedLesson(
60                               JOptionPane.showInputDialog("First name:"),
61                               JOptionPane.showInputDialog("Last name:"),
62                               JOptionPane.showInputDialog("Lesson:")
63                       );
64                       break;
65   
66                   case 4:
67                       tracker.viewAllStudents();
68                       break;
69   
70                   case 5:
71                       tracker.saveToFile();
72                       break;
73   
74                   case 6:
75                       tracker.loadFromFile();
76                       break;
77   
78                   case 7:
79                       tracker.deleteStudent(
80                               JOptionPane.showInputDialog("First name:"),
81                               JOptionPane.showInputDialog("Last name:")
82                       );
83                       break;
84   
85                   case 8:
86                       tracker.viewStudent(
87                               JOptionPane.showInputDialog("First name:"),
88                               JOptionPane.showInputDialog("Last name:")
89                       );
90                       break;
91   
92                   case 9:
93                       tracker.editStudent(
94                               JOptionPane.showInputDialog("Enter current first name:"),
95                               JOptionPane.showInputDialog("Enter current last name:")
96                       );
97                       break;
98   
99                   case 10:
100                      JOptionPane.showMessageDialog(null, "Tracker closed.");
101                      return;
102  
103                  default:
104                      JOptionPane.showMessageDialog(null, "Invalid option.");
105              }
106          }
107      }
108  }
